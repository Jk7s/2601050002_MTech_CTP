import datetime

bookedslots = {
    22: {"vehicle": "2W", "entry": datetime.datetime.now()},
    34: {"vehicle": "4W", "entry": datetime.datetime.now()},
    87: {"vehicle": "2W", "entry": datetime.datetime.now()},
    12: {"vehicle": "4W", "entry": datetime.datetime.now()},
    3: {"vehicle": "2W", "entry": datetime.datetime.now()}
}


def show_slots():
    for i in range(1, 101, 10):
        for j in range(i, i + 10):
            if j in bookedslots:
                print("*", end="\t")
            else:
                print(j, end="\t")
        print()

    print("Available slots:", 100 - len(bookedslots))


def book_slot():
    if len(bookedslots) == 100:
        print("Sorry, parking area is full.")
        return

    print("1. Two Wheeler")
    print("2. Four Wheeler")

    choice = int(input("Enter vehicle type: "))

    if choice == 1:
        vehicle = "2W"
    elif choice == 2:
        vehicle = "4W"
    else:
        print("Invalid vehicle type")
        return

    slot = int(input("Choose slot to book: "))

    if slot < 1 or slot > 100:
        print("Invalid slot number")
    elif slot in bookedslots:
        print("This slot is already booked.")
    else:
        bookedslots[slot] = {
            "vehicle": vehicle,
            "entry": datetime.datetime.now()
        }

        print("Slot booked successfully")
        print("Entry time:", bookedslots[slot]["entry"])


def release_slot():
    slot = int(input("Enter slot number: "))

    if slot not in bookedslots:
        print("Slot is already empty.")
        return

    entry = bookedslots[slot]["entry"]
    vehicle = bookedslots[slot]["vehicle"]

    exit_time = datetime.datetime.now()

    hours = (exit_time - entry).total_seconds() / 3600
    hours = max(1, int(hours) + (1 if hours % 1 > 0 else 0))

    if vehicle == "2W":
        charge = hours * 20
    else:
        charge = hours * 40

    print("Exit time:", exit_time)
    print("Parking time:", hours, "hour(s)")
    print("Parking charge: ₹", charge)

    del bookedslots[slot]

    print("Slot released successfully")


def parking_system():
    while True:
        print("\n1. Show Slots")
        print("2. Book Slot")
        print("3. Release Slot")
        print("4. Exit")

        choice = int(input("Enter choice: "))

        if choice == 1:
            show_slots()
        elif choice == 2:
            book_slot()
        elif choice == 3:
            release_slot()
        elif choice == 4:
            break
        else:
            print("Invalid choice")


parking_system()
